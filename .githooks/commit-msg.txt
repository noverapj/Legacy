#!/bin/sh

COMMIT_MSG_FILE="$1"
COMMIT_MSG=$(head -n 1 "$COMMIT_MSG_FILE")

PATTERN='^(feat|fix|refactor|perf|docs|test|build|chore|revert)\((client|engine|gamesvr|mainsvr|billingsvr|dbagent|filewritesvr|loginsvr|relaysvr|launcher|tools|sql|build|deps|docs)\): .+'

if echo "$COMMIT_MSG" | grep -Eq "$PATTERN"; then
    exit 0
fi

echo ""
echo "ERROR: Invalid commit message."
echo ""
echo "Expected:"
echo "  type(scope): description"
echo ""
echo "Valid types:"
echo "  feat fix refactor perf docs test build chore revert"
echo ""
echo "Valid scopes:"
echo "  client engine gamesvr mainsvr billingsvr dbagent"
echo "  filewritesvr loginsvr relaysvr launcher tools sql"
echo "  build deps docs"
echo ""
echo "Examples:"
echo "  feat(client): add new hero skill handler"
echo "  fix(gamesvr): prevent duplicate room join"
echo "  fix(engine): preserve binary ABI for RenderSkip"
echo ""

exit 1
